  Widget _buildMostPopularTrackCard(Map<String, dynamic> artistData) {
    String trackName = 'Unknown Track';
    double streams = 0;
    String? imageUrl;
    
    // Find most popular track
    if (artistData['topTracks'] != null && (artistData['topTracks'] as List).isNotEmpty) {
      final track = (artistData['topTracks'] as List).first as Map;
      trackName = track['title'] ?? track['name'] ?? 'Unknown Track';
      streams = _parseNum(track['streamCount']).toDouble();
      if (streams == 0) streams = _parseNum(track['playcount']).toDouble();
      imageUrl = track['thumbnail'] ?? track['artwork'] ?? (artistData['images'] != null && (artistData['images'] as List).isNotEmpty ? ((artistData['images'] as List)[0] as Map)['url'] : null);
    }

    return Container(
      padding: const EdgeInsets.all(20),
      decoration: BoxDecoration(
        color: Colors.white.withValues(alpha: 0.05),
        borderRadius: BorderRadius.circular(24),
      ),
      child: Row(
        children: [
          if (imageUrl != null)
            ClipRRect(
              borderRadius: BorderRadius.circular(12),
              child: Image.network(imageUrl, width: 64, height: 64, fit: BoxFit.cover),
            )
          else
            Container(width: 64, height: 64, decoration: BoxDecoration(color: Colors.grey.shade800, borderRadius: BorderRadius.circular(12))),
          const SizedBox(width: 16),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text('MOST POPULAR TRACK', style: GoogleFonts.inter(color: Colors.white54, fontSize: 10, fontWeight: FontWeight.bold, letterSpacing: 1)),
                const SizedBox(height: 4),
                Text(trackName, maxLines: 1, overflow: TextOverflow.ellipsis, style: GoogleFonts.inter(color: Colors.white, fontSize: 18, fontWeight: FontWeight.bold)),
                const SizedBox(height: 4),
                Row(
                  children: [
                    const Icon(Icons.play_circle_fill, color: Color(0xFF1DB954), size: 16),
                    const SizedBox(width: 4),
                    Text('${_formatNumber(streams)} Streams', style: GoogleFonts.inter(color: Colors.white, fontSize: 14, fontWeight: FontWeight.bold)),
                  ],
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }

  double _extractStat(Map<String, dynamic> artistData, List<String> keys) {
    final stats = artistData['stats'] as Map?;
    for (final key in keys) {
      if (stats != null && stats[key] != null) return _parseNum(stats[key]).toDouble();
      if (artistData[key] != null) return _parseNum(artistData[key]).toDouble();
    }
    return 0;
  }

  num _parseNum(dynamic value) {
    if (value == null) return 0;
    if (value is num) return value;
    if (value is String) {
      // Remove any commas or text that might break parsing
      final cleanStr = value.replaceAll(RegExp(r'[^0-9.-]'), '');
      return num.tryParse(cleanStr) ?? 0;
    }
    return 0;
